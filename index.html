const SPREADSHEET_ID = "10UXiZCWcBw7fH3CFTqnYINT8AnRO0YIxwZJCymypcFw"; const DRIVE_FOLDER_ID = "1dkXORKv5qI7Dmik6HQgpxmcIGUg3wcu3"; function doGet(e) { let action = e && e.parameter && e.parameter.action ? e.parameter.action : ""; let callback = e && e.parameter && e.parameter.callback ? e.parameter.callback : ""; let result = { success: false, message: "Aksi tidak dikenal" }; try { setupDatabaseAutomatically(); if (action === "getExpenseOptions") { result = { success: true, data: getExpenseOptionsData() }; } else if (action === "getExpenses") { result = { success: true, data: getAllExpenses() }; } else { result = { success: true, message: "API Expense Tracker Aktif" }; } } catch (err) { result = { success: false, message: err.toString() }; } if (callback) { let jsonString = JSON.stringify(result); return ContentService.createTextOutput(callback + "(" + jsonString + ")") .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { try { setupDatabaseAutomatically(); let data = JSON.parse(e.postData.contents); let action = data.action; let payload = data.payload; let result = { success: false, message: "Aksi tidak dikenal" }; if (action === "saveExpense") { result = saveExpenseRecord(payload); } return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ success: false, message: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } } function setupDatabaseAutomatically() { try { let ss = SpreadsheetApp.openById(SPREADSHEET_ID); let sheet = ss.getSheetByName("Pengeluaran"); if (!sheet) { sheet = ss.insertSheet("Pengeluaran"); sheet.appendRow([ "Timestamp", "ID Transaksi", "Tanggal", "Nama Karyawan", "Kategori Biaya", "Nominal Biaya", "Keterangan", "Bukti Transaksi (URL)" ]); sheet.getRange("1:1").setFontWeight("bold"); sheet.setFrozenRows(1); } return sheet; } catch (err) { throw new Error("Gagal melakukan setup database otomatis: " + err.toString()); } } function getExpenseSheet() { return setupDatabaseAutomatically(); } function getExpenseOptionsData() { return { kategori: ["Perizinan", "Makan & minum", "Transportasi", "Percetakan", "Telepon & internet", "Lainnya"], karyawan: ["Sugi Puswanto"] }; } function saveExpenseRecord(payload) { let sheet = getExpenseSheet(); let folder; try { folder = DriveApp.getFolderById(DRIVE_FOLDER_ID); } catch (e) { folder = DriveApp.getRootFolder(); } let fileUrls = []; if (payload.files && Array.isArray(payload.files)) { payload.files.forEach((fileObj, index) => { if (fileObj.base64 && fileObj.name) { try { let parts = fileObj.base64.split(";base64,"); let contentType = parts[0].replace("data:", ""); let rawData = Utilities.base64Decode(parts[1]); let blob = Utilities.newBlob(rawData, contentType, "Nota_" + payload.id_transaksi + "_" + (index + 1) + "_" + fileObj.name); let savedFile = folder.createFile(blob); savedFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); fileUrls.push(savedFile.getUrl()); } catch (err) { fileUrls.push("Gagal Upload: " + err.toString()); } } }); } let timestamp = new Date(); sheet.appendRow([ timestamp, // A: Timestamp payload.id_transaksi, // B: ID Transaksi payload.tanggal, // C: Tanggal payload.nama_karyawan, // D: Nama Karyawan payload.kategori, // E: Kategori Biaya payload.nominal, // F: Nominal Biaya payload.keterangan, // G: Keterangan fileUrls.join(", ") // H: Bukti Transaksi (URL) ]); return { success: true, message: "Pengeluaran berhasil disimpan ke Google Spreadsheet!" }; } function getAllExpenses() { let sheet = getExpenseSheet(); let rows = sheet.getDataRange().getValues(); let list = []; for (let i = 1; i < rows.length; i++) { list.push({ timestamp: rows[i][0], id_transaksi: rows[i][1], tanggal: rows[i][2], nama_karyawan: rows[i][3], kategori: rows[i][4], nominal: rows[i][5], keterangan: rows[i][6], bukti: rows[i][7] }); } return list; }
